```java
public void moveZeroes(int[] nums) {

    for(int fast=0,slow=0;fast<nums.length;fast++){
	    if(nums[fast]!=0){
	    temp=nums[slow];
	    nums[slow]=nums[fast];
	    nums[fast]=temp;
	    slow++;
	    }
    }
}
```
